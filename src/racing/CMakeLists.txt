set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(native_lane_pilot STATIC src/pilot.cpp)
target_include_directories(native_lane_pilot PUBLIC src)
target_link_libraries(native_lane_pilot PUBLIC camera control track)
add_library(race_vehicle_core STATIC src/mission_main.cpp ../tool/race_backend_loader.cpp)
target_include_directories(race_vehicle_core PUBLIC tasks src ../tool ../detection/include)
target_link_libraries(race_vehicle_core PUBLIC native_lane_pilot camera control track joystick nlohmann_json pthread dl)
set_target_properties(detection PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(race_npu_backend SHARED ../tool/race_perception.cpp)
target_include_directories(race_npu_backend PRIVATE tasks src ../tool)
target_link_libraries(race_npu_backend PRIVATE detection nlohmann_json pthread)
target_link_options(race_npu_backend PRIVATE -static-libstdc++ -static-libgcc)
if(CAR_CXX_RUNTIME)
  target_link_options(race_npu_backend PRIVATE "-L${CAR_CXX_RUNTIME}/lib")
endif()
add_executable(racing src/main.cpp)
target_link_libraries(racing PRIVATE race_vehicle_core)
add_dependencies(racing race_npu_backend)
